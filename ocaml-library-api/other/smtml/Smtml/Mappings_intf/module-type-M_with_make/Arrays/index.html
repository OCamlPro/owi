Module M_with_make.Arrays

val select : term -> term -> term

select a i retrieves the element of array a at index i.

val store : term -> term -> term -> term

store a i v returns the array obtained by associating the index i to the element v in the array a.

val const : (ty -> term -> term) option

const index_ty v is the constant array with index type index_ty that maps every index to v, or None if the solver can't build constant arrays.
